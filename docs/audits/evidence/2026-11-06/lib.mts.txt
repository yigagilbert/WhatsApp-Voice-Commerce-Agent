export type Ev = { t: number; label: string; ms: number; kind: string };
export const events: Ev[] = [];
let origin = 0;
export const mark = () => { origin = performance.now(); events.length = 0; };
export function record(kind: string, label: string, started: number) {
  events.push({ t: started - origin, label, ms: performance.now() - started, kind });
}
export function timedDb() {
  const base = createPrismaClient({ databaseUrl: process.env.DATABASE_URL! });
  return base.$extends({ query: { $allModels: { async $allOperations({ model, operation, args, query }: any) {
    const s = performance.now();
    try { return await query(args); } finally { record('db', `${model}.${operation}`, s); }
  } } } });
}
export class TimedModel {
  id: string; inner: any; usage: any[] = [];
  constructor(inner: any) { this.inner = inner; this.id = inner.id; }
  async complete(req: any) {
    const s = performance.now();
    const r = await this.inner.complete(req);
    record('llm', `llm(msgs=${req.messages.length},tools=${req.tools?.length ?? 0})`, s);
    this.usage.push({ ms: performance.now() - s, ...(r.usage ?? {}), promptChars: JSON.stringify(req.messages).length, toolChars: JSON.stringify(req.tools ?? []).length, toolCalls: r.message.toolCalls?.map((c: any) => c.name) ?? [] });
    return r;
  }
}
