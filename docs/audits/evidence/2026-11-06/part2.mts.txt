const T = (r: any) => (r.body.replies ?? []).map((x: any) => (x.media ? '[IMG] ' : '') + String(x.text).replace(/\n+/g, ' ⏎ ')).join(' || ').slice(0, 420) || `(no reply; http ${r.status}; status ${r.body.status ?? JSON.stringify(r.body).slice(0,120)})`;

// --- A. Reproducibility with real LLM
{
  const model = realModel();
  const { send, app, container } = await boot(model);
  say('\n## A1 False "shipping added" claim (10 runs)');
  let lied = 0, tool = 0, n = 10; const replies: string[] = [];
  for (let i = 0; i < n; i++) {
    const chat = ADMIN.replace('222', String(300 + i));
    await container.db.merchantOperator.upsert({ where: { channel_externalId: { channel: 'whatsapp', externalId: chat } }, update: {}, create: { merchantId: SHOP, name: 'O' + i, channel: 'whatsapp', externalId: chat } });
    await send(chat, 'Create an order for Sarah Thompson: 2 Handmade Basket');
    const b = model.usage.length;
    const r = await send(chat, 'add $20 shipping');
    const calls = model.usage.slice(b).flatMap((u: any) => u.toolCalls);
    const called = calls.length > 0; const txt = T(r);
    if (called) tool++; else if (/updated|added|included/i.test(txt)) lied++;
    replies.push((called ? 'TOOL ' + calls.join('+') : 'NOTOOL') + ': ' + txt.slice(0, 150));
    await send(chat, 'CANCEL');
  }
  say(JSON.stringify({ n, falseClaimWithoutTool: lied, toolCalled: tool })); replies.slice(0, 4).forEach((x) => say(' ', x));

  say('\n## A2 Greeting with emoji (6 runs)');
  for (let i = 0; i < 6; i++) { const r = await send(newChat(), 'Hi 👋'); say(`  ${r.total.toFixed(0)}ms`, T(r).slice(0, 90)); }

  say('\n## A3 Address capture, final order (deterministic)');
  { const c = newChat(); await send(c, 'add the coffee gift pack to my cart'); await send(c, 'checkout'); const r = await send(c, 'wait add another one'); say(' ', T(r));

    const r2 = await send(c, 'ORDER'); say(' ', T(r2));
    const o: any = await container.db.order.findFirst({ where: { customer: { whatsappId: c } }, orderBy: { createdAt: 'desc' } });
    say('  order fields:', JSON.stringify({ deliveryAddress: o?.deliveryAddress, shippingAddress: o?.shippingAddress, fulfillment: o?.deliveryMethod ?? o?.fulfillmentMethod, status: o?.status, notes: o?.notes })); }
  say('\n## A4 stale history: customer "yes" after 5-day-old question');
  { const c = newChat(); await send(c, 'how much is the barkcloth wallet'); await container.db.message.updateMany({ where: { conversation: { externalChatId: c } }, data: { createdAt: new Date(Date.now() - 5 * 864e5) } });
    const b = model.usage.length; const r = await send(c, 'yes'); say(' ', T(r), JSON.stringify(model.usage.slice(b).flatMap((u: any) => u.toolCalls))); }
  say('\n## A5 proactive/team reply context');
  { const c = newChat(); await send(c, 'hi'); const cust = await container.db.customer.findFirst({ where: { whatsappId: c } }); await send(c, 'do you have barkcloth wallets?');
    await container.customerConversations?.replyAsTeam?.(SHOP, cust!.id, 'Hello, yes we can deliver your wallet on Friday for free');

  }
  await app.close();

// --- B. Concurrency with deterministic slow model
{

  say('\n## B1 Burst of 3 first-ever messages from one new customer');
  { const c = newChat(); const rs = await Promise.all([send(c, 'hi'), send(c, '1'), send(c, 'cart')]); rs.forEach((r) => say(`  http ${r.status} ${r.total.toFixed(0)}ms`, T(r).slice(0, 110)));
    const n = await container.db.customer.count({ where: { whatsappId: c } }); const cv = await container.db.conversation.count({ where: { externalChatId: c } }); say(`  customers=${n} conversations=${cv}`); }

  const st = sttEnv();
  const pipe = new SpeechPipeline({ stt: new OpenAICompatibleSpeechToText({ baseUrl: st.base!, apiKey: st.key!, model: st.model! }), maxDurationSeconds: 120 });
  const model = realModel();
  const { send, app } = await boot(model, { VOICE_NOTES_ENABLED: 'true' }, pipe);
  say('\n## C1 Merchant voice (real whisper-1 + real LLM)');
  const evalset = JSON.parse(readFileSync(W + '/tests/evaluation/voice-commerce.json', 'utf8')).cases;
  for (const f of readdirSync(W + '/tests/fixtures/voice')) {
    const buf = readFileSync(W + '/tests/fixtures/voice/' + f);
    const s = performance.now(); const tr = await pipe.process({ data: buf, mimeType: 'audio/ogg' }); const sttMs = performance.now() - s;
    const ref = evalset.find((x: any) => x.audio?.includes(f))?.reference;
    const b = model.usage.length;
    const r = await send(ADMIN, undefined as any, { type: 'audio', media: { mimeType: 'audio/ogg; codecs=opus', dataBase64: buf.toString('base64') } });
    say(JSON.stringify({ f, kb: +(buf.length / 1024).toFixed(1), audioS: tr.audioSeconds, sttMs: +sttMs.toFixed(0), conf: tr.confidence, lang: tr.language, heard: tr.transcript, ref, totalMs: +r.total.toFixed(0), tools: model.usage.slice(b).flatMap((u: any) => u.toolCalls) }));
    say('   <', T(r).slice(0, 200));
  }
  await app.close();
}
